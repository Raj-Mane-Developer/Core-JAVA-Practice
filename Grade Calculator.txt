 public class Grade_Calculator {

	public static void main(String[] args) {

		Scanner scan = new Scanner(System.in);
		
		 double marks = scan.nextDouble() ;
			if (marks>=37 && marks <=49.999)
			{
			System.out.print("D grade");
			}

			else if (marks>=50 && marks <=59.999)
			{
			System.out.print("C grade");

			}

			else if(marks>=60 && marks<=69.999)
			{
			System.out.print("B grade");
			}
			else if (marks>=70 && marks<=79.999)
			{
			System.out.print("A grade");

			}

			else if (marks>=80 && marks<=100)
			{
			System.out.print("MERIT");

			}
			else if (marks<37 && marks<=0 )
			{
		 		System.out.println("FAIL");
			}
			else
			{
			System.out.println("INVALID");
			System.out.println("Please Give Valid Percentage");

			}


		
		
	}
